import numpy as np

def square_partial_sum(x, N):
    """Sum the first N odd harmonics."""
    n = np.arange(1, 2 * N, 2)   # 1, 3, 5, …
    terms = np.sin(np.outer(x, n)) / n
    return (4 / np.pi) * terms.sum(axis=1)
